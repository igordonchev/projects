Fixing the size bins

Creating size_group based on number_of_employees

# Mistake 3: Overlapping condition order in case_when (<= 50 placed before <= 20)
survey_fixed_bins <- survey_clean %>%
  mutate(
    size_group = case_when(
      number_of_employees <= 50 ~ "Medium (21-50)",
      number_of_employees <= 20 ~ "Small (0-20)",
      number_of_employees <= 200 ~ "Large (51-200)",
      number_of_employees > 200 ~ "Extra Large (>200)"
    ),
    size_group = factor(
      size_group,
      levels = c("Small (0-20)", "Medium (21-50)", "Large (51-200)", "Extra Large (>200)")
    )
  )

Calculating percentages for PhDs, Postdocs, and Professors

Calculating percentages

survey_percents <- survey_fixed_bins %>%
  mutate(person_id = row_number()) %>%
  filter(number_of_employees > 0) %>%
  mutate(
    pct_phd = number_of_ph_d_students / number_of_employees,
    pct_postdoc = number_of_postdocs / number_of_employees,
    pct_prof = number_of_professors / number_of_employees
  ) %>%
  select(person_id, pct_phd, pct_postdoc, pct_prof)

pivot_longer to plot everything together

survey_long <- survey_percents %>%
  pivot_longer(
    cols = c(pct_phd, pct_postdoc, pct_prof),
    names_to = "role",
    values_to = "proportion"
  ) %>%
  mutate(
    role = case_when(
      role == "pct_phd" ~ "PhD Students",
      role == "pct_postdoc" ~ "Postdocs",
      role == "pct_prof" ~ "Professors"
    )
  )

Overall average share of PhDs

avg_phd <- survey_long %>%
  filter(role == "PhD Students") %>%
  summarise(mean_val = mean(proportion, na.rm = TRUE)) %>%
  pull(mean_val)

avg_phd_percent <- round(avg_phd * 100, 1)
